import UIKit
import Onnxruntime

class DocScanner {
    private var session: ORTSession?
    private let inputSize = 288

    func init_(modelPath: String) throws {
        let env = ORTEnv()
        let options = ORTSessionOptions()
        options.graphOptimizationLevel = .all

        guard let session = try? ORTSession(
            env: env,
            modelPath: modelPath,
            sessionOptions: options
        ) else {
            throw NSError(domain: "DocScanner", code: 1, userInfo: [NSLocalizedDescriptionKey: "Failed to create session"])
        }
        self.session = session
    }

    func scan(image: UIImage) -> UIImage? {
        guard let cgImage = image.cgImage else { return nil }

        // Preprocess
        guard let inputTensor = preprocess(cgImage: cgImage) else { return nil }

        // Inference
        guard let session = session else { return nil }
        guard let inputName = session.inputNames.first else { return nil }
        guard let outputName = session.outputNames.first else { return nil }

        do {
            let outputs = try session.run(
                [outputName],
                inputs: [inputName: inputTensor]
            )

            guard let outputData = outputs.first else { return nil }

            // Get flow field from output
            let flow = try outputData.tensorData.withUnsafeBytes { ptr -> [[Float]] in
                let floatPtr = ptr.bindMemory(to: Float.self)
                var result = [[Float]](repeating: [Float](repeating: 0, count: 288), count: 288)
                for y in 0..<288 {
                    for x in 0..<288 {
                        let idx = y * 288 + x
                        result[y][x] = floatPtr[idx]
                    }
                }
                return result
            }

            return applyWarp(image: image, flowX: flow)
        } catch {
            print("Inference error: \(error)")
            return nil
        }
    }

    private func preprocess(cgImage: CGImage) -> ORTValue? {
        let width = inputSize
        let height = inputSize

        // Resize and convert to float array
        var floatData = [Float](repeating: 0, count: 3 * height * width)

        guard let context = CGContext(
            data: &floatData,
            width: width,
            height: height,
            bitsPerComponent: 8,
            bytesPerRow: width * 4,
            space: CGColorSpaceCreateDeviceRGB(),
            bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue
        ) else { return nil }

        context.draw(cgImage, in: CGRect(x: 0, y: 0, width: width, height: height))

        // Convert RGB to BGR and normalize to [0, 1]
        var chwData = [Float](repeating: 0, count: 3 * height * width)
        for i in 0..<(height * width) {
            let r = floatData[i * 4] / 255.0
            let g = floatData[i * 4 + 1] / 255.0
            let b = floatData[i * 4 + 2] / 255.0
            chwData[i] = b
            chwData[height * width + i] = g
            chwData[2 * height * width + i] = r
        }

        let shape: [NSNumber] = [1, 3, height, width]
        return try? ORTValue(
            tensorData: Data(bytes: chwData, count: chwData.count * MemoryLayout<Float>.size),
            shape: shape,
            elementType: .float
        )
    }

    private func applyWarp(image: UIImage, flowX: [[Float]]) -> UIImage? {
        guard let cgImage = image.cgImage else { return nil }

        let width = cgImage.width
        let height = cgImage.height

        guard let context = CGContext(
            data: nil,
            width: width,
            height: height,
            bitsPerComponent: 8,
            bytesPerRow: width * 4,
            space: CGColorSpaceCreateDeviceRGB(),
            bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue
        ) else { return nil }

        context.draw(cgImage, in: CGRect(x: 0, y: 0, width: width, height: height))

        guard let data = context.data else { return nil }

        let pixels = data.bindMemory(to: UInt8.self, capacity: width * height * 4)

        for y in 0..<height {
            for x in 0..<width {
                let fx = Int(((flowX[y][x] / 143.4) + 1) * Float(width - 1) / 2)
                let fy = Int(((flowX[y][x + height * width] / 143.4) + 1) * Float(height - 1) / 2)

                let srcX = max(0, min(width - 1, fx))
                let srcY = max(0, min(height - 1, fy))

                let dstIdx = (y * width + x) * 4
                let srcIdx = (srcY * width + srcX) * 4

                pixels[dstIdx] = pixels[srcIdx]
                pixels[dstIdx + 1] = pixels[srcIdx + 1]
                pixels[dstIdx + 2] = pixels[srcIdx + 2]
                pixels[dstIdx + 3] = 255
            }
        }

        guard let newCGImage = context.makeImage() else { return nil }
        return UIImage(cgImage: newCGImage)
    }
}
