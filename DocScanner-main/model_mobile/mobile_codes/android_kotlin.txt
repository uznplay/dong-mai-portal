package com.docscanner

import ai.onnxruntime.*
import android.graphics.Bitmap
import android.graphics.BitmapFactory
import java.nio.FloatBuffer

class DocScanner {
    private var session: OrtSession? = null
    private var env: OrtEnvironment? = null

    fun init(modelPath: String = "docscanner.onnx") {
        env = OrtEnvironment.getEnv()
        val sessOpts = OrtSession.SessionOptions()
        sessOpts.graphOptimizationLevel = OrtSession.SessionOptions.GRAPH_OPTIMIZATION_LEVEL_ALL
        session = env!!.createSession(modelPath, sessOpts)
    }

    fun scan(bitmap: Bitmap): Bitmap {
        val input = preprocess(bitmap)
        val output = inference(input)
        return postprocess(bitmap, output)
    }

    private fun preprocess(bitmap: Bitmap): FloatBuffer {
        val resized = Bitmap.createScaledBitmap(bitmap, 288, 288, true)
        val buf = FloatBuffer.allocate(3 * 288 * 288)

        for (y in 0 until 288) {
            for (x in 0 until 288) {
                val p = resized.getPixel(x, y)
                buf.put(((p shr 16) and 0xFF) / 255f)  // R
                buf.put(((p shr 8) and 0xFF) / 255f)   // G
                buf.put((p and 0xFF) / 255f)           // B
            }
        }
        buf.rewind()
        return buf
    }

    private fun inference(buf: FloatBuffer): Array<Array<FloatArray>> {
        val inputName = session!!.inputNames.first()
        val outputName = session!!.outputNames.first()

        val shape = longArrayOf(1, 3, 288, 288)
        val tensor = OnnxTensor.createTensor(env!!, buf, shape)

        val outputs = session!!.run(mapOf(inputName to tensor))
        return (outputs[0].value as Array<Array<FloatArray>>)
    }

    private fun postprocess(orig: Bitmap, flow: Array<Array<FloatArray>>): Bitmap {
        val w = orig.width
        val h = orig.height
        val result = Bitmap.createBitmap(w, h, Bitmap.Config.ARGB_8888)

        for (y in 0 until h) {
            for (x in 0 until w) {
                val srcX = ((flow[0][y][x] / 143.4f + 1f) * (w - 1) / 2f).toInt().coerceIn(0, w - 1)
                val srcY = ((flow[1][y][x] / 143.4f + 1f) * (h - 1) / 2f).toInt().coerceIn(0, h - 1)
                result.setPixel(x, y, orig.getPixel(srcX, srcY))
            }
        }
        return result
    }
}
